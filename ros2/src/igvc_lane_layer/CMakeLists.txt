cmake_minimum_required(VERSION 3.8)
project(igvc_lane_layer)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rclcpp REQUIRED)
add_library(lane_layer SHARED src/lane_layer.cpp)
target_compile_features(lane_layer PUBLIC cxx_std_17)
ament_target_dependencies(lane_layer nav2_costmap_2d pluginlib sensor_msgs rclcpp)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS lane_layer LIBRARY DESTINATION lib)
ament_package()
